package jogo_de_trap;

import javax.swing.*;
import java.awt.*;
import java.awt.event.*;

public class StartPanel extends JPanel {
    private JFrame frame;
    
    // Variáveis de controle do Hover/Seleção
    boolean startBtnHover = false;
    private ModoJogo modoSelecionado = ModoJogo.NORMAL;
    private boolean[] modoHover = new boolean[3]; // 0=Facil, 1=Normal, 2=Dificil
    private final ModoJogo[] valoresModo = {ModoJogo.FACIL, ModoJogo.NORMAL, ModoJogo.DIFICIL};
    
    // Componente de texto para atualizar a descrição
    private JTextArea description;

    // Coordenadas para cálculo do desenho (baseado no seu layout)
    private final int PANEL_X = 44;
    private final int PANEL_Y = 185;
    private final int MODES_X = 25;
    private final int MODES_Y = 166;
    private final int BTN_WIDTH = 74; // 224 dividido por 3
    private final int BTN_HEIGHT = 43;

    public StartPanel(JFrame frame) {
        this.frame = frame;
        this.setLayout(null);

        // --- Configuração do Botão Start Principal (Mantido) ---
        ImageIcon resizedIcon = new ImageIcon(
                new ImageIcon(getClass().getResource("/assets/telas_e_botoes/btnStart.png"))
                        .getImage().getScaledInstance(180, 70, Image.SCALE_SMOOTH));

        this.addKeyListener(new KeyAdapter() {
            @Override
            public void keyPressed(KeyEvent e) {
                if (e.getKeyCode() == KeyEvent.VK_SPACE) iniciarJogo();
            }
        });

        JButton startButton = new JButton(resizedIcon);
        startButton.setOpaque(false);
        startButton.setContentAreaFilled(false);
        startButton.setBorderPainted(false);
        startButton.setFocusPainted(false);
        startButton.setBounds(25, 33, 224, 48);

        startButton.addActionListener(e -> iniciarJogo());
        startButton.addMouseListener(new MouseAdapter() {
            public void mouseEntered(MouseEvent e) {
                startButton.setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));
                startBtnHover = true;
                repaint();
            }
            public void mouseExited(MouseEvent e) {
                startButton.setCursor(Cursor.getPredefinedCursor(Cursor.DEFAULT_CURSOR));
                startBtnHover = false;
                repaint();
            }
        });
        this.add(startButton);

        // --- Painel Interno ---
        JPanel panel = new JPanel();
        panel.setBounds(PANEL_X, PANEL_Y, 280, 363);
        panel.setLayout(null);
        panel.setOpaque(false); // Importante para ver o background desenhado no pai
        add(panel);

        JButton btnIniciar = new JButton("Iniciar"); // Placeholder visual
        btnIniciar.setBounds(25, 33, 224, 48);
        panel.add(btnIniciar);

        JButton btnConfig = new JButton("Configurações"); // Placeholder visual
        btnConfig.setBounds(25, 102, 224, 43);
        panel.add(btnConfig);

        // --- Lógica dos 3 Botões de Modo (Substituindo o btnModes) ---
        String[] labels = {"Fácil", "Normal", "Difícil"};
        
        for (int i = 0; i < 3; i++) {
            final int index = i;
            JButton btn = new JButton(labels[i]);
            // Posiciona um ao lado do outro
            btn.setBounds(MODES_X + (i * BTN_WIDTH), MODES_Y, BTN_WIDTH, BTN_HEIGHT);
            
            // Estilo invisível (desenharemos no paintComponent)
            btn.setOpaque(false);
            btn.setContentAreaFilled(false);
            btn.setBorderPainted(false);
            btn.setFocusPainted(false);
            btn.setForeground(Color.WHITE); // Cor do texto do botão

            btn.addMouseListener(new MouseAdapter() {
                public void mouseEntered(MouseEvent e) {
                    modoHover[index] = true;
                    repaint();
                }
                public void mouseExited(MouseEvent e) {
                    modoHover[index] = false;
                    repaint();
                }
                public void mouseClicked(MouseEvent e) {
                    modoSelecionado = valoresModo[index];
                    description.setText(modoSelecionado.descricao); // Atualiza texto
                    repaint();
                }
            });
            panel.add(btn);
        }

        description = new JTextArea();
        description.setBounds(21, 236, 228, 89);
        description.setOpaque(false); // Transparente ou com cor de fundo
        description.setForeground(Color.WHITE);
        description.setLineWrap(true);
        description.setWrapStyleWord(true);
        description.setText(modoSelecionado.descricao); // Texto inicial
        description.setEditable(false);
        panel.add(description);
    }

    private void iniciarJogo() {
        // Lógica de iniciar (adicionar verificação do modoSelecionado se precisar passar pro jogo)
        System.out.println("Iniciando jogo no modo: " + modoSelecionado);
        // ... sua lógica de troca de tela ...
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);

        // 1. Background Image
        ImageIcon icon = new ImageIcon(getClass().getResource("/assets/telas_e_botoes/StartBG.jpg"));
        Image bg = icon.getImage();
        
        // Sua lógica de escala
        int imgWidth = bg.getWidth(null);
        int imgHeight = bg.getHeight(null);
        int panelWidth = this.getWidth();
        double scaleFactor = (double) panelWidth / imgWidth;
        int scaleAltura = (int) (imgHeight * scaleFactor) - 110;
        
        g.drawImage(bg, 0, 0, panelWidth, scaleAltura, null);

        Graphics2D g2d = (Graphics2D) g.create();
        
        // 2. Outline do Botão Start (Lógica antiga)
        if (startBtnHover) {
            g2d.setColor(Color.WHITE);
            g2d.setStroke(new BasicStroke(3));
            g2d.drawRect(300, 450, 180, 70); // Ajuste conforme sua coordenada real se mudou
        }

        // 3. Outline dos Botões de Modo (NOVA LÓGICA)
        // Precisamos somar a posição do 'panel' (44, 185) com a posição dos botões relativos (25, 166)
        int absoluteRefX = PANEL_X + MODES_X; 
        int absoluteRefY = PANEL_Y + MODES_Y;

        for (int i = 0; i < 3; i++) {
            if (modoHover[i] || modoSelecionado == valoresModo[i]) {
                Color c = (modoSelecionado == valoresModo[i]) ? Color.CYAN : Color.WHITE; // Cyan p/ selecionado fica bonito no tema espacial
                g2d.setColor(c);
                g2d.setStroke(new BasicStroke(2));
                
                // Desenha o retângulo na posição calculada
                g2d.drawRect(
                    absoluteRefX + (i * BTN_WIDTH), 
                    absoluteRefY, 
                    BTN_WIDTH, 
                    BTN_HEIGHT
                );
            }
        }
        g2d.dispose();
    }
}
